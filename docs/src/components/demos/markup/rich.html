                richPayload reads the computed styles, writes them inline, turns every colour into hex, and
                leaves out scripts, controls, hidden content and tracking images. Links and images become absolute.
              
    
              
    
                In dark mode, the card keeps its own dark background, so its light text stays readable. The paragraph
                below has no background of its own: on a white email its light text would vanish, so
                colors: "readable" leaves it to the destination's colour.
              
    
              
    
                
    import { richPayload, richTransform } from 'react-smart-copy/rich';

copy(() => richPayload(cardRef.current));

useCopyInterceptor(articleRef, { transform: richTransform() });
