      Objects in, objects out
      
              
      
                clipPayload writes the cards three ways: a web custom format, a custom clipboard type and a
                hidden envelope in the HTML. The paste button, Ctrl + V and drops each read a
                different one, so every path in every browser finds the cards.
              
      
              
      
                Whatever is pasted is checked before your code sees it: size-limited, parsed without prototype keys,
                migrated from older versions, then validated. "Copy a tampered clip" writes the same format with bad
                data, the way a buggy or hostile app could.
              
      
              
      
                
      import { clipAccept, clipPayload, defineClipFormat, readClip } from 'react-smart-copy/structured';

const Cards = defineClipFormat({
  id: 'acme.cards',
  version: 2,
  schema: z.array(Card), // or Valibot, ArkType, a function
  migrate: { 1: (cards) => addTags(cards) },
});

copy(() => clipPayload(Cards, selected));

usePaste({ accept: clipAccept(Cards), onPaste: async (result) => {
  const clip = await readClip(result, Cards);
  if (clip.status === 'ok') addToBacklog(clip.value);
} });

              Pasting
              
          
              
            Three paths: a button that calls paste() (may prompt for permission),
            targetProps.onPaste for keyboard paste, and dropTargetProps for
            drag-and-drop. The last two need no prompt, because the data arrives in the event, and
            all three go through the same accept, limits and image checks.
          
              
          
              
            
              import { usePaste } from 'react-smart-copy';

function PasteZone() {
  const { paste, targetProps, dropTargetProps, isDragOver, status, result } = usePaste({
    accept: ['text/plain', 'image/png'],
  });

  return (
    <>
      <button type="button" onClick={() => void paste()}>
        {status === 'reading' ? 'Reading…' : 'Paste'}
      </button>
      <div {...targetProps} {...dropTargetProps} tabIndex={0} data-over={isDragOver || undefined}>
        Drop here, or Ctrl+V
      </div>
      {result?.text && <pre>{result.text}</pre>}
    </>
  );
}

              Retry and cancellation
              
          
              
            When the Paste button fails for a reason that can pass (permission denied, page not
            focused), retry() reads again, up to maxRetries, and
            canRetry says whether it can help. Every copy and paste also carries an
            AbortSignal that fires on reset(), on unmount, or when a newer
            operation takes over, so slow captures and large decodes stop instead of running on.
            onCancel(reason) tells you when that happens, with 'reset',
            'disconnect' or, for paste, 'superseded'; onReset()
            runs after reset() clears a result.
          
              
          
              
            
              const { paste, retry, canRetry } = usePaste({
  accept: ['image'],
  onCancel: (reason) => analytics.track('paste_cancelled', { reason }),
});

// Image sources and custom adapters receive the signal:

              Controlling native copy
              
          
              
            useCopyInterceptor decides what reaches the clipboard when someone selects text inside an element
            and presses Ctrl/⌘+C, Ctrl/⌘+X or uses the context menu. Return a string or payload to write instead,
            false to block, or nothing to leave the browser alone. Try it in the playground.
          
              
          
              
            
              import { useCopyInterceptor } from 'react-smart-copy';

function Credentials({ children }) {
  const { ref } = useCopyInterceptor<HTMLDivElement>({
    priority: 10, // wins selections that also touch other scopes
    transform: ({ fragment, text }) => {
      if (!fragment) return '[redacted]'; // inside an <input>
      fragment.querySelectorAll('[data-redact]').forEach((n) => (n.textContent = '••••'));
      const box = document.createElement('div');
      box.append(fragment);
      return { kind: 'html', html: box.innerHTML, text: box.textContent };
    },
    onError: (error) => report(error), // the copy is blocked: fails closed
  });
  return <div ref={ref}>{children}</div>;
}
              
          
              
          
              
            Writes go through the copy event, so there is no permission prompt and no HTTPS requirement. Only strings
            can be written that way: the types reject image payloads and async transforms. Cuts in inputs and
            editable regions still remove the selection, with undo intact. When several scopes overlap, one handles the
            selection: highest priority, then innermost, then first in the page. Without React, use
            registerCopyInterceptor(element, options) from react-smart-copy/core.
          
              
          
              It stops accidental leaks, not determined users: screenshots and dev tools still work.
              
        
              

        
              
          
              Drag to copy
              
          
              
            useDragCopy makes an element draggable and decides what lands wherever it is dropped: a text
            editor, a rich editor, a spreadsheet or your own drop zone. Return a string or payload, false to
            cancel the drag, or nothing to keep the browser's default. Try it in the playground.
          
              
          
              
            
              import { useDragCopy } from 'react-smart-copy';

function InvoiceCard({ invoice }) {
  const { ref, status } = useDragCopy<HTMLDivElement>({
    source: () => ({
      kind: 'html',
      html: `<b>${invoice.id}</b>, ${invoice.client}`,
      text: `${invoice.id}, ${invoice.client}`,
    }),
    effectAllowed: 'copy',
    enabled: !invoice.locked,
  });
  return <div ref={ref} data-status={status}>{invoice.id}</div>;
}
              
          
              
          
              
            Data goes into the drag's DataTransfer, so there is no Clipboard API, no permission prompt and no
            HTTPS requirement. source runs synchronously on dragstart and always sees the latest
            props. Text, HTML, JSON and multi payloads work; images don't, because DataTransfer only takes
            strings. Without React, use registerDragCopy(element, options) from
            react-smart-copy/core.

              Structured data
              
          
              
            react-smart-copy/structured copies your app's own objects and pastes them back as the same
            typed, validated objects in another tab, window or app: rows, widgets, form configurations, shapes.
            Text editors and mail get readable text instead. Try it in the playground.
          
              
          
              
            
              import { clipAccept, clipPayload, defineClipFormat, readClip } from 'react-smart-copy/structured';

export const InvoiceRows = defineClipFormat({
  id: 'acme.invoice-rows',           // namespaced: it names the clipboard type
  version: 2,
  schema: z.array(Invoice),          // any Standard Schema, or a function that throws
  migrate: { 1: (rows) => upgradeFromV1(rows) },
  text: (rows) => rows.map((row) => row.id).join('\n'),
});

copy(() => clipPayload(InvoiceRows, rows));   // also from a copy interceptor or drag source

usePaste({ accept: [...clipAccept(InvoiceRows), 'text'], onPaste: async (result) => {
  const clip = await readClip(result, InvoiceRows);
  // clip.status: 'ok' | 'absent' | 'invalid' | 'unsupported-version'
} });
              
          
              
          
              
            No single clipboard entry survives every path in every browser, so each copy carries the data three ways
            and readClip takes the first that holds the format: the web custom format (the paste button in
            Chromium), the custom type (Ctrl + V and drops after a native copy or drag) and a hidden
            envelope in the HTML (every path, including Safari and Firefox).
          
              
          
              
            Pasted data is untrusted: each carrier is size-checked before it is parsed, __proto__ keys are
            dropped, migrations run, then the schema validates. Reading never throws for clipboard content. The data is
            not authenticated, so check permissions on the server, and copy only what the user may share: the HTML
            envelope travels with the HTML (turn it off with embedInHtml: false).
          
              
        
              

        
              
          
              Paste as Markdown

            into clean Markdown, from Google Docs, Word, Notion, GitHub, web pages and mail.
            Try it in the playground.

<textarea value={body} onChange={…} onPaste={pasteAsMarkdown} />

              Copy and paste code
              
          
              
            react-smart-copy/code is for the copy button on every code block and for fields people paste
            code into. Try it in the playground.
          
              
          
              
            
              import { cleanCode, codePayload, inspectCode, pasteAsCode, pasteToCode } from 'react-smart-copy/code';

// A rendered, highlighted block: commands without prompts, no gutters, colours kept as HTML.
copy(() => codePayload(preRef.current, { language: 'tsx' }));
copy(() => codePayload(diffText, { diff: 'after' }));

// A code field: cleaned, dedented, indented to the cursor, undo intact.
<textarea onPaste={pasteAsCode} />

// Anywhere, including the server.
inspectCode(source).danger;   // Trojan Source, invisible identifiers, lookalike letters
cleanCode(source).code;       // safe replacements, applied until nothing is left
              
          
              
          
              
            A terminal or REPL session (bash, zsh, PowerShell, cmd, Python, IPython, Node) copies as its commands.
            Continued commands stay whole: a trailing backslash, a heredoc or an open quote. A unified diff copies as
            either side, and typed line numbers are removed when they count up.
          
              
          
              
            inspectCode reports each finding with its line, column and Unicode name. It covers bidi
            overrides and isolates (CVE-2021-42574), Hangul fillers that make invisible identifiers, tag characters
            that hide ASCII, runs of variation selectors, Cyrillic and Greek lookalikes inside Latin words, control
            characters, unusual spaces and typographic punctuation. Joiners in Arabic, Persian and Indic words and in
            emoji, and full-width punctuation in CJK comments, are left alone. Pasted code is never cut: past
            maxLength the paste is left to the browser.
          
              
        
              

        
              
          
              Pasted images: private and small
              
          
              
            react-smart-copy/image gets pasted and dropped images ready to upload: the location, camera
            data and hidden previews are removed, and the image fits your limits. Try it in the playground.
          
              
          
              
            
              import { prepareImage, readImageInfo, stripMetadata } from 'react-smart-copy/image';

// In the browser: strip, resize and compress under a budget; re-encodes only when it has to.
const image = await prepareImage(file, { maxWidth: 2048, maxBytes: 1_000_000 });
upload(image.file);

// Anywhere, including the server: no decoding, the same pixels.
const info = await readImageInfo(file);   // { format, width, height, orientation, metadata: ['exif', 'gps', …] }
const { blob, removed } = await stripMetadata(file);
              
          
              
          
              
            JPEG, PNG, WebP and GIF are rewritten without Exif, GPS, thumbnails, XMP, IPTC, comments, text chunks,
            multi-picture and depth images, content credentials and appended data. The colour profile and the
            orientation are kept. AVIF and HEIC are re-encoded. Sizes are checked before decoding. A byte budget
            searches quality first, then size, and is a too-large error when it can't be met. Every parser
            read is bounded, and fuzz tests feed it random and mutated files.

                  The payload for copy(), the interceptor and drag-copy: web custom format, custom type, HTML envelope and text.
                
              
                
              
                  readClip(source, format), readClipSync, clipAccept(...formats)
                  /structured
                  Read a clip from a paste result, DataTransfer or payload: ok, absent, invalid or unsupported-version.
                
              
                
              
                  pasteAsMarkdown(event, options?)
                  /markdown
                  A paste handler for Markdown fields: rich pastes become Markdown at the cursor, with undo intact; plain text is left alone.
                
              
                
              
                  htmlToMarkdown(source, options?), pasteToMarkdown(result)
                  /markdown
                  HTML string or DOM node to CommonMark + GFM; Markdown for a paste result. { markdown, truncated }.
                
              
                
              
                  insertText(field, text), escapeMarkdown(text)
                  /markdown
                  Insert at the cursor keeping native undo; make any text render literally.
                
              
                
              
                  codePayload(source, options?), codeFromElement(element), normalizeCode(text)
                  /code
                  Copy code from a string or a rendered block: prompts and output, gutters and buttons left out; text plus highlighted HTML.
                
              
                
              
                  pasteAsCode(event, options?), pasteToCode(result)
                  /code
                  Paste code cleaned, dedented and indented to the cursor, with undo intact. { code, findings, fixed, remaining, danger }.
                
              
                
              
                  inspectCode(text), cleanCode(text)
                  /code
                  Find, or safely replace, bidi overrides, invisible and control characters, odd spaces, curly punctuation and lookalike letters.
                
              
                
              
                  prepareImage(file, options?)
                  /image
                  Strip, resize and compress an image for upload; lossless when only metadata goes. { file, width, height, size, reencoded, removed }.
                
              
                
              
                  stripMetadata(file), readImageInfo(file)
                  /image
                  Remove metadata without re-encoding; read format, size, orientation and metadata from the bytes. Server-safe.
                
              
                
              
                  registerDragCopy(element, options)
                  /core
                  Framework-agnostic drag-to-copy. Returns { unsubscribe }; applyDragCopy(event, options) handles a single event.

                  /capture
                  SVG element or markup string → PNG Blob. No external dependency.

                  /capture
                  DOM element → PNG Blob via caller-supplied rasteriser.
                
              
                
              
                  captureImage(getNode, options?)
                  /capture
